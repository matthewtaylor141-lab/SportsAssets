        teleSub.textContent=(pct>=0?'+':'−')+Math.abs(pct).toFixed(0)+'% vs prev 30d';
        teleSub.className='tele-sub '+(pct>=0?'up':'down'); }
      else { teleSub.textContent='vs prev 30d · no data'; teleSub.className='tele-sub'; }
    }
    var days=r.per_day||[]; if(!teleBars||!days.length) return;
    var max=0; days.forEach(function(d){ if(d.count>max) max=d.count; });
    if(telePeak){ var p=compact(max); telePeak.textContent='peak '+p[0]+p[1]; }
    if(teleBars.children.length!==days.length){
      teleBars.innerHTML=days.map(function(d,i){
        return '<i style="animation-delay:'+(i*0.03).toFixed(2)+'s"'+(i===days.length-1?' class="now"':'')+'></i>';
      }).join('');
    }
    for(var i=0;i<days.length;i++) teleBars.children[i].style.height=(max?6+Math.round(days[i].count/max*70):6)+'px';
  }
  function renderStatus(feed){
    if(!stTable) return;
    if(stLive) stLive.textContent=fmt(feed.live_events);
    if(stPre)  stPre.textContent=fmt(feed.prematch_events);
    if(stUpd)  stUpd.textContent='updated '+new Date().toTimeString().slice(0,8);
    var lv=(feed.live_by_sport||[]), pm=(feed.prematch_by_sport||[]);
    if(!lv.length&&!pm.length) return; // endpoint degraded — keep the waiting state
    var live={}, pre={};
    lv.forEach(function(s){ live[s.sport_id]=s.count; });
    pm.forEach(function(s){ pre[s.sport_id]=s.count; });
    stTable.innerHTML=SPORTS.map(function(sp){
      var l=live[sp[0]]||0, p=pre[sp[0]]||0;
      return '<div class="st-cell"><span class="st-ic">'+sp[2]+'</span><span class="st-nm">'+sp[1]+'</span>'
        +'<span class="st-n"><b'+(l?'':' class="zero"')+'>'+fmt(l)+'</b><i>·</i>'
        +'<span'+(p?'':' class="zero"')+'>'+fmt(p)+'</span></span></div>';
    }).join('');
  }
  function renderSports(list){
    if(!list||!list.length){ barsEl.innerHTML='<div class="ls-empty">No drops yet today — markets are quiet. Check back during games.</div>'; return; }
    var max=Math.max.apply(null,list.map(function(s){return s.count;}))||1;
    barsEl.innerHTML=list.slice(0,10).map(function(s){
      var w=Math.max(4,Math.round(s.count/max*100));
      return '<div class="ls-bar"><span class="ls-bar-name">'+esc(s.name||'Other')+'</span>'
        +'<span class="ls-bar-track"><span class="ls-bar-fill" style="width:'+w+'%"></span></span>'
        +'<span class="ls-bar-n">'+fmt(s.count)+'</span></div>';
    }).join('');
  }
  function load(){
    fetch('/panel/api/public-stats',{cache:'no-store'}).then(function(r){return r.json();}).then(function(d){
      var feed=d.feed||{};
      dropsEl.textContent=fmt(d.drops_today);
      liveEl.textContent=fmt(feed.live_events);
      preEl.textContent=fmt(feed.prematch_events);
      if(botsEl) botsEl.textContent=fmt((+feed.ws_clients_connected||0)+(+feed.sse_clients||0)+(+feed.prematch_sse_clients||0));
      renderSports(d.drops_today_by_sport);
      renderStatus(feed);
      renderTele(d.requests_30d);
    }).catch(function(){});
  }
  load(); setInterval(load, 15000);
})();

  var reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* ---------- scroll-reveal ---------- */
  var els = document.querySelectorAll('.reveal');
  if (reduce || !('IntersectionObserver' in window)) {
    els.forEach(function(e){ e.classList.add('in'); });
  } else {
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(en){
        if (en.isIntersecting){ en.target.classList.add('in'); io.unobserve(en.target); }
      });
    }, { threshold: 0.15, rootMargin: '0px 0px -8% 0px' });
    els.forEach(function(e){ io.observe(e); });
  }

  /* ---------- nav stays pinned (always sticky) ---------- */

  var burger = document.getElementById('burger'), navEl = document.getElementById('nav');
  if (burger && navEl) {
    burger.addEventListener('click', function(){ navEl.classList.toggle('open'); });
    // tapping a menu link closes the dropdown
    navEl.querySelectorAll('.nav-links a').forEach(function(a){
      a.addEventListener('click', function(){ navEl.classList.remove('open'); });
    });
  }

  /* ---------- feature card: live mini price-chart ----------
     A rolling Pinnacle price line that ticks and drops — on-theme with
     "event-driven detection". Renders + updates regardless of reduced-motion
     (it's product content), and is never empty. */
  (function miniChart(){
    var box = document.getElementById('bigviz');
    if (!box) return;
    var W=600, H=120, N=48;
    box.innerHTML = '<svg class="pl-svg" viewBox="0 0 '+W+' '+H+'" preserveAspectRatio="none">'
      + '<polygon class="pl-area" points=""/>'
      + '<polyline class="pl-line" points=""/>'
      + '<circle class="pl-dot" r="3.2"/></svg>';
    var line = box.querySelector('.pl-line'),
        area = box.querySelector('.pl-area'),
        dot  = box.querySelector('.pl-dot');
    var lo=1.55, hi=2.45, p=2.0, data=[], t=0, dropHold=0;
    for (var i=0;i<N;i++) data.push(p);
    function yfor(v){ return (H-8) - ((v-lo)/(hi-lo))*(H-22); }
    function xfor(i){ return (i/(N-1))*W; }
    function draw(){
      var pts='', j;
      for (j=0;j<N;j++){ pts += (j?' ':'') + xfor(j).toFixed(1)+','+yfor(data[j]).toFixed(1); }
      line.setAttribute('points', pts);
      area.setAttribute('points', '0,'+H+' '+pts+' '+W+','+H);
      dot.setAttribute('cx', W-1);
      dot.setAttribute('cy', yfor(data[N-1]).toFixed(1));
    }
    function stepChart(){
      t++;
      var isDrop = (t % 7 === 0);
      if (isDrop){
        p = Math.max(lo+0.06, p - (0.10 + Math.abs(Math.sin(t))*0.13));
        box.classList.add('drop'); dropHold = 3;
      } else {
        p = p + Math.sin(t*1.3)*0.045 + (2.0 - p)*0.05; // wander + mean-revert
        if (dropHold>0 && --dropHold===0) box.classList.remove('drop');
      }
      p = Math.min(hi-0.04, Math.max(lo+0.04, p));
      data.push(+p.toFixed(3)); if (data.length>N) data.shift();
      draw();
    }
    draw();
    setInterval(stepChart, 560);
  })();

  /* ---------- referral capture (?ref=) ---------- */
  // Persist a referral code so signup can attribute it. Mirrors the legacy flow.
  try { var _rp = new URL(location.href).searchParams.get('ref'); if (_rp) sessionStorage.setItem('po_ref', _rp.slice(0,32)); } catch(e){}
  function getStoredRef(){ try { return sessionStorage.getItem('po_ref') || null; } catch(e){ return null; } }

  /* getAcq() reads the first-touch record written by /acq.js (loaded below). */
  function getAcq(){ return window.poAcq ? window.poAcq() : {}; }
  /* getDevice() asks /ui-hints.js for the browser's device id; computed only now, at signup. */
  async function getDevice(){ try { return window.poDevice ? await window.poDevice() : {}; } catch(e){ return {}; } }
  async function jpost(url, body){
    try { var r = await fetch(url, { method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify(body) });
          return { status:r.status, data: await r.json().catch(function(){ return null; }) }; }
    catch(e){ return { status:0, data:null }; }
  }

  /* ---------- signup modal (wired to /panel/api/signup) ---------- */
  function closeModal(){ document.querySelectorAll('.modal-bg').forEach(function(n){ n.remove(); }); }
  function showSignupModal(planLabel){
    closeModal();
    var paid = !!planLabel;
    var heading = paid ? 'Create your account' : 'Get your free trial key';
    var intro = paid
      ? 'You\'re subscribing to <b style="color:var(--text)">' + planLabel + '</b>. First create your free account — enter your email, click the verification link, and you\'ll continue straight to checkout. The key we issue is your login (no password).'
      : 'Enter your email — we\'ll send a verification link. Click it and your API key appears on the next screen. The key is your login; there\'s no password.';
    var goText = paid ? 'Continue' : 'Send verification email';
    var bg = document.createElement('div');
    bg.className = 'modal-bg';
    bg.innerHTML =
      '<div class="modal" role="dialog" aria-modal="true">'
      + '<button class="m-x" type="button" aria-label="Close">×</button>'
      + '<h3>' + heading + '</h3>'
      + '<p>' + intro + '</p>'
      + '<input id="suEmail" type="email" placeholder="you@example.com" autocomplete="email"/>'
      + '<div class="m-err" id="suErr"></div>'
      + '<button class="btn btn-primary m-go" id="suGo">' + goText + '</button>'
      + '<div class="m-sub">'
      +   '<button class="m-link" id="suCancel">Cancel</button>'
      +   '<a class="m-link" href="/panel">I already have a key →</a>'
      + '</div>'
      + '<p class="m-note">Lost your key? <a href="/panel" style="color:var(--accent)">Recover it →</a></p>'
      + '</div>';
    document.body.appendChild(bg);
    bg.addEventListener('click', function(e){ if (e.target === bg) closeModal(); });
    var emailEl = document.getElementById('suEmail');
    var errEl = document.getElementById('suErr');
    var goBtn = document.getElementById('suGo');
    emailEl.focus();
    emailEl.addEventListener('keydown', function(e){ if (e.key === 'Enter') goBtn.click(); });
    document.getElementById('suCancel').onclick = closeModal;
    goBtn.onclick = async function(){
      errEl.textContent = '';
      var email = emailEl.value.trim();
      if (!email){ errEl.textContent = 'Enter an email.'; return; }
      goBtn.disabled = true; goBtn.textContent = 'Sending…';
      var r = await jpost('/panel/api/signup', Object.assign({ email: email, ref: getStoredRef() }, getAcq(), await getDevice()));
      goBtn.disabled = false; goBtn.textContent = goText;
      if (r.status !== 200){ errEl.textContent = (r.data && r.data.message) || ('Signup failed: ' + ((r.data && r.data.error) || r.status)); return; }
      // Google Ads conversion — fires only on a successful (200) email submit, not on resend/error
      try {
        if (typeof gtag === 'function') gtag('event', 'conversion', {
          'send_to': 'AW-18328425741/3e5wCPXQv9EcEI2q1qNE',
          'value': 1.0,
          'currency': 'EUR',
          'transaction_id': ''
        });
      } catch (e) {}
      showCheckEmailModal(email, planLabel);
    };
  }
  function showCheckEmailModal(email, planLabel){
    closeModal();
    var safe = String(email).replace(/[<>&]/g, '');
    var planNote = planLabel
      ? '<p style="color:var(--text);font-size:13px;margin-top:12px">After you verify, your dashboard has a one-click button to finish checkout for <b>' + planLabel + '</b>. Your trial key works in the meantime.</p>'
      : '';
    var bg = document.createElement('div');
    bg.className = 'modal-bg';
    bg.innerHTML =
      '<div class="modal" role="dialog" aria-modal="true" style="text-align:center">'
      + '<button class="m-x" type="button" aria-label="Close">×</button>'
      + '<div class="m-emoji">📬</div>'
      + '<h3>Check your email</h3>'
      + '<p>We sent a verification link to <b style="color:var(--text)">' + safe + '</b>.<br>Click it and your API key appears on the next screen.</p>'
      + planNote
      + '<p class="m-note">Link expires in 24h. No email? Check Spam/Promotions, or <a href="#" id="suResend" style="color:var(--accent)">resend</a>.</p>'
      + '<div class="m-row" style="justify-content:center;margin-top:16px"><button class="btn btn-ghost" id="suClose">Close</button></div>'
      + '</div>';
    document.body.appendChild(bg);
    bg.addEventListener('click', function(e){ if (e.target === bg) closeModal(); });
    document.getElementById('suClose').onclick = closeModal;
    document.getElementById('suResend').onclick = async function(e){
      e.preventDefault(); e.target.textContent = 'resending…';
      await jpost('/panel/api/signup', Object.assign({ email: email, ref: getStoredRef() }, getAcq(), await getDevice()));
      e.target.textContent = 'sent again ✓'; e.target.style.pointerEvents = 'none';
    };
  }
  document.addEventListener('keydown', function(e){ if (e.key === 'Escape') closeModal(); });
  document.addEventListener('click', function(e){ if (e.target && e.target.classList && e.target.classList.contains('m-x')) closeModal(); });

  /* ---------- wire signup CTAs ----------
     Any .js-signup opens the signup modal. Paid-plan CTAs stash the chosen plan
     (so the panel offers one-click checkout after verification) and pass its
     label so the modal reflects what they're subscribing to.
     ("Sign in" / "Dashboard" keep their plain /panel link.) */
  var PLAN_LABELS = {
    stream_30d:'Drops', stream_90d:'Drops',
    pro_30d:'REST', pro_90d:'REST', pro_180d:'REST',
    pro_stream_30d:'Edge', pro_stream_90d:'Edge', pro_stream_180d:'Edge',
    scale_30d:'Scale', scale_90d:'Scale'
  };
  // Billing matrix — keep in sync with saas/plans.js. Some tiers aren't offered
  // on every cycle (Drops & Scale have no semi-annual); missing cells render as
  // unavailable. WS add-on surcharge scales with term ($99 * days/30).
  var PLANS_MATRIX = {
    stream:     { monthly:{sku:'stream_30d',price:99,days:30,disc:0}, quarterly:{sku:'stream_90d',price:249,days:90,disc:16} },
    pro:        { monthly:{sku:'pro_30d',price:99,days:30,disc:0}, quarterly:{sku:'pro_90d',price:249,days:90,disc:16}, semi:{sku:'pro_180d',price:449,days:180,disc:24} },
    pro_stream: { monthly:{sku:'pro_stream_30d',price:149,days:30,disc:0}, quarterly:{sku:'pro_stream_90d',price:379,days:90,disc:15}, semi:{sku:'pro_stream_180d',price:679,days:180,disc:24} },
    scale:      { monthly:{sku:'scale_30d',price:229,days:30,disc:0}, quarterly:{sku:'scale_90d',price:599,days:90,disc:13} },
  };
  var currentPeriod = 'monthly';
  function updateCard(card){
    var tier = card.getAttribute('data-tier'); if (!tier) return;
    var info = (PLANS_MATRIX[tier] || {})[currentPeriod];
    var cta = card.querySelector('.js-signup');
    var amtEl = card.querySelector('.price-amt');
    var perEl = card.querySelector('.price-per');
    var subEl = card.querySelector('.price-sub');
    var wsOpt = card.querySelector('.ws-opt');
    var wsCb = card.querySelector('.ws-opt-toggle');
    if (!info){
      if (amtEl) amtEl.textContent = '—';
      if (perEl) perEl.textContent = '';
      if (subEl) subEl.textContent = 'Only on monthly / quarterly';
      if (wsOpt) wsOpt.style.display = 'none';
      if (cta){ cta.classList.add('disabled'); cta.removeAttribute('data-plan'); }
      return;
    }
    if (cta){ cta.classList.remove('disabled'); cta.setAttribute('data-plan', info.sku); }
    var wsSur = Math.round(99 * info.days / 30);
    var wsOn = !!(wsCb && wsCb.checked);
    var total = info.price + (wsOn ? wsSur : 0);
    if (amtEl) amtEl.textContent = total;
    if (perEl) perEl.textContent = (currentPeriod === 'monthly') ? '/mo' : ('/' + info.days + 'd');
    if (subEl){
      if (currentPeriod === 'monthly'){ subEl.textContent = ''; }
      else {
        var perMo = Math.round(total / (info.days / 30));
        subEl.textContent = '= $' + perMo + '/mo' + (info.disc ? (' · save ' + info.disc + '%') : '');
      }
    }
    if (wsOpt){
      wsOpt.style.display = '';
      var wp = wsOpt.querySelector('.ws-opt-price');
      if (wp) wp.textContent = 'add $' + wsSur + (currentPeriod === 'monthly' ? '/mo' : '');
    }
  }
  function updateAllCards(){ document.querySelectorAll('.plan[data-tier]').forEach(updateCard); }
  document.querySelectorAll('.ws-opt-toggle').forEach(function(cb){
    cb.addEventListener('change', function(){ var c = cb.closest('.plan'); if (c) updateCard(c); });
  });
  document.querySelectorAll('#billToggleLanding button').forEach(function(b){
    b.addEventListener('click', function(){
      document.querySelectorAll('#billToggleLanding button').forEach(function(x){ x.classList.remove('active'); });
      b.classList.add('active');
      currentPeriod = b.getAttribute('data-period');
      updateAllCards();
    });
  });
  updateAllCards();
  document.querySelectorAll('.js-signup').forEach(function(el){
    el.addEventListener('click', function(e){
      e.preventDefault();
      if (el.classList.contains('disabled')) return;

      if (p && p !== 'trial'){
        var pending = { planId:p, ts:Date.now() };
        var card = el.closest('.plan');
        var cb = card ? card.querySelector('.ws-opt-toggle') : null;
        if (cb && cb.checked) pending.addons = ['ws'];

  /* ====================================================================
     LIVE TERMINAL ENGINE  — hybrid: real SSE when configured, else a
     self-contained simulated loop so the page is always alive.
     SAFETY: never embed a customer/admin API key here. LIVE_SSE.token is
     intentionally empty in the public page → simulated mode. An operator
     can wire a public read-only token at deploy time to go truly live.
     HONESTY: prices/drops below are illustrative (clearly labelled). The
     latency band (15–40ms frame→bot) and the race (push vs a 1s poller)
     reflect real product behaviour; we never claim to beat ps3838.
     ==================================================================== */
  var LIVE_SSE = { url: '/odds-drop', token: '' };  // token empty ⇒ simulated demo

  var feedEl    = document.getElementById('tlFeed');
  var latEl     = document.getElementById('tlLat');
  var latA = document.getElementById('latPulseA'), latB = document.getElementById('latPulseB');
  var segA = document.getElementById('latSegA'),  segB = document.getElementById('latSegB');
  if (!feedEl) return; // hero not present

  // Real-style Pinnacle markets (Team v Team · market). Shown verbatim;
  // when the delayed real feed kicks in, real drops replace these.
  var MARKETS = [
    { nm:'Man City v Arsenal · ML Home',       p:1.874 },
    { nm:'Real Madrid v Bayern · AH -0.5',     p:1.952 },
    { nm:'Knicks v Timberwolves · O 221.5',    p:1.886 },
    { nm:'Sinner v Alcaraz · Set 2 ML',        p:1.742 },
    { nm:'Oilers v Avalanche · PL -1.5',       p:2.184 },
    { nm:'Chiefs v Bills · U 47.5',            p:1.906 },
    { nm:'Lakers v Celtics · O 224.5',         p:1.628 },
    { nm:'OSFP v PAO · BTTS Yes',              p:1.722 },
    { nm:'Liverpool v Newcastle · ML Away',    p:4.310 },
    { nm:'Djokovic v Rune · Game Hcp -3.5',    p:1.833 },
    { nm:'Inter v Milan · DNB Home',           p:1.694 },
    { nm:'Dodgers v Padres · RL -1.5',         p:2.052 }
  ];
  var REAL = false; // flips true once the delayed public feed has real drops

  function fmt(x){ return (+x).toFixed(3); }
  function hhmmss(ts){
    var d = new Date(ts || Date.now());
    function z(n){ return (n < 10 ? '0' : '') + n; }
    return z(d.getHours()) + ':' + z(d.getMinutes()) + ':' + z(d.getSeconds());
  }

  // prepend a feed row (newest first), cap at 7. The "caught today" counter
  // is NOT bumped here — it shows the real daily total from the stats poll.
  function pushRow(nm, from, to, pct, ts){
    var row = document.createElement('div');
    row.className = 'tl-row new';
    row.innerHTML = '<span class="tl-ts">' + hhmmss(ts) + '</span>'
      + '<span class="tl-mkt">' + nm + '</span>'
      + '<span class="tl-px">' + fmt(from) + ' <i>→</i> <b>' + fmt(to) + '</b></span>'
      + '<span class="tl-pct">-' + pct.toFixed(2) + '%</span>';
    feedEl.insertBefore(row, feedEl.firstChild);
    while (feedEl.children.length > 7) feedEl.removeChild(feedEl.lastChild);
  }

  var latTick = 0;
  // Rerolls ONCE PER DROP (never on a timer) so the ms only changes when a new
  // row lands. With a live feed, pass the REAL measured total (now − frame
  // timestamp); otherwise a realistic 15–40 ms value stands in.
  function runPulse(el){ if (!el) return; el.classList.remove('go'); void el.offsetWidth; el.classList.add('go'); }
  function rerollLat(realTotal){
    latTick++;
    var ms;
    if (realTotal != null && realTotal > 1 && realTotal < 2000){
      ms = Math.round(realTotal);                            // genuine frame → your bot
    } else {
      var h = (Math.sin(latTick * 91.317) * 4271.73) % 1;    // pseudo-random, non-periodic
      ms = 15 + Math.floor(Math.abs(h) * 26);                // 15–40 ms
    }
    // split the total across the two hops: Pinnacle → pinnapi, pinnapi → your bot
    var h2 = Math.abs((Math.sin(latTick * 53.971) * 2917.41) % 1);
    var a = Math.max(3, Math.round(ms * (0.38 + h2 * 0.14))), b = Math.max(1, ms - a);
    if (latEl) latEl.textContent = ms;
    if (segA) segA.textContent = a + ' ms';
    if (segB) segB.textContent = b + ' ms';
    runPulse(latA);                                  // signal runs Pinnacle → pinnapi …
    setTimeout(function(){ runPulse(latB); }, 320);  // … then pinnapi → your bot
  }

  var dropN = 0;
  // one illustrative drop: generate a row + reroll the latency readout
  function dropOne(ts){
    if (REAL) return;            // real drops come from the delayed feed instead
    dropN++;
    var m  = MARKETS[(dropN * 5 + 3) % MARKETS.length];      // 5 ⟂ 12 → shuffled full cycle
    var h1 = Math.abs((Math.sin(dropN * 53.971) * 2917.41) % 1);
    var h2 = Math.abs((Math.sin(dropN * 17.313) * 7919.77) % 1);
    var from = +(m.p * (0.96 + h1 * 0.09)).toFixed(3);       // jitter around the base price
    var pct  = 0.8 + h2 * 3.6;                               // illustrative 0.8–4.4% moves
    var to   = +(from * (1 - pct / 100)).toFixed(3);
    pushRow(m.nm, from, to, pct, ts);
    rerollLat();                 // ms updates only when a drop lands
  }

  /* ---------- speed race ---------- */
  var fillYou = document.getElementById('fillYou');
  var fillThem= document.getElementById('fillThem');
  var runYou  = document.getElementById('runYou');
  var runThem = document.getElementById('runThem');
  var timeYou = document.getElementById('timeYou');
  var timeThem= document.getElementById('timeThem');
  var raceDelta = document.getElementById('raceDelta');
  var raceN = 0;

  function setLane(fill, runner, frac, dur){
    if (fill){ fill.style.transition='none'; fill.style.width='0'; }
    if (runner){ runner.style.transition='none'; runner.style.left='0'; }
    void (fill && fill.offsetWidth);
    requestAnimationFrame(function(){
      if (fill){ fill.style.transition='width '+dur+'ms cubic-bezier(.23,1,.32,1)'; fill.style.width=(frac*100)+'%'; }
      if (runner){ runner.style.transition='left '+dur+'ms cubic-bezier(.23,1,.32,1)'; runner.style.left='calc('+(frac*100)+'% - 8px)'; }
    });
  }

  function runRace(){
    raceN++;
    var youMs  = 16 + Math.floor(Math.abs(Math.sin(raceN*1.9))*22);   // 16–38ms
    var themMs = 430 + Math.floor(Math.abs(Math.cos(raceN*1.3))*520); // 430–950ms (poll staleness)
    // visual durations (scaled so it's watchable; labels show the real ms)
    setLane(fillYou, runYou, 1, 900);
    setLane(fillThem, runThem, 1, Math.min(2600, 900*(themMs/youMs)/6));
    if (timeYou)  timeYou.textContent  = youMs + 'ms';
    if (timeThem) timeThem.textContent = themMs + 'ms';
    if (raceDelta) raceDelta.innerHTML = 'You caught it <b>' + (themMs - youMs) + 'ms</b> earlier';
  }

  /* ---------- boot: simulated or real SSE ---------- */
  function startSimulated(){
    // prime a full feed (back-dated timestamps) so it's never empty on load
    for (var i = 6; i >= 0; i--) dropOne(Date.now() - i * 2200);
    setInterval(dropOne, 2200);  // a new row every 2.2s — the ms rerolls per drop
    runRace();
    setInterval(runRace, 4000);  // re-run the speed race
    startRealFeed();             // upgrade to real (delayed) drops when available
  }

  /* ---------- real (delayed) public feed ----------
     Pulls REAL Pinnacle drops held back 10s (so it can't be traded live) and
     lets them drive the feed. Until real drops arrive it stays on the
     synthetic demo above; once it goes real it doesn't flip back. */
  var seenReal = {};
  // The footer copy ("Real Pinnacle drops, delayed 10 s …") is static in the
  // markup now — this only flips the data source from synthetic to real.
  function setReal(on){ REAL = on; }
  function applyReal(drops){
    if (!drops || !drops.length) return;
    setReal(true);
    // feed = drops we haven't shown yet, oldest first so newest ends on top
    var fresh = [];
    for (var j = 0; j < drops.length; j++){
      var key = drops[j].ts + '|' + drops[j].nm;
      if (seenReal[key]) continue; seenReal[key] = 1; fresh.push(drops[j]);
    }
    fresh.sort(function(a,b){ return a.ts - b.ts; });
    fresh.forEach(function(d){
      pushRow(d.nm, +d.from, +d.to, Math.abs(+d.pct || 0), d.ts);
      rerollLat();
    });
  }
  function startRealFeed(){
    function poll(){
      fetch('/api/public-feed', { cache:'no-store' })
        .then(function(r){ return r.ok ? r.json() : null; })
        .then(function(j){ if (j && j.drops) applyReal(j.drops); })
        .catch(function(){});
    }
    poll(); setInterval(poll, 5000);
  }

  function startLive(){
    // Operator opt-in: real /odds-drop stream with a public read-only token.
    try {
      var es = new EventSource(LIVE_SSE.url + '?token=' + encodeURIComponent(LIVE_SSE.token));
      var gotData = false, fellBack = false;
      var guard = setTimeout(function(){ if (!gotData && !fellBack){ fellBack=true; try{es.close();}catch(e){} startSimulated(); } }, 6000);
      es.onmessage = function(ev){
        gotData = true;
        var arr; try { arr = JSON.parse(ev.data); } catch(e){ return; }
        (Array.isArray(arr)?arr:[arr]).forEach(function(d){
          if (!d || d.to_price == null) return;
          // Real match name + price, plus the REAL measured latency (now − ts).
          var pct = d.drop_pct!=null ? +d.drop_pct : ((d.from_price-d.to_price)/d.from_price*100);
          var realMs = d.ts ? (Date.now() - d.ts) : null;   // genuine frame → your bot
          pushRow((d.market||d.event||'market'), +d.from_price, +d.to_price, Math.abs(pct), d.ts);
          rerollLat(realMs); runRace();
        });
      };
      es.onerror = function(){ if (!gotData && !fellBack){ fellBack=true; clearTimeout(guard); try{es.close();}catch(e){} startSimulated(); } };
    } catch(e){ startSimulated(); }
  }

  if (LIVE_SSE.token) startLive(); else startSimulated();
})();
</script>
